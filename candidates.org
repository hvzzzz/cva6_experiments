re_begin_npb = re.compile(r">>>BEGIN_TEST\|NPB_(\w+)_")
re_begin_unix = re.compile(r">>>BEGIN_TEST\|(\w+)")
re_end = re.compile(r">>>END_TEST")

# Metrics
re_mops = re.compile(r"\s*Mop/s total\s*=\s*([\d\.]+)")
re_unix_count = re.compile(r"COUNT\|([\d\.]+)\|(\d+)\|(\w+)")
re_papi_inst = re.compile(r"\[PAPI\] Instructions:\s*(\d+)")
re_papi_cyc = re.compile(r"\[PAPI\] Cycles:\s*(\d+)")
re_papi_ipc = re.compile(r"\[PAPI\] IPC:\s*([\d\.]+)")

def parse_logs(file_path, suite_type):
    if not os.path.exists(file_path):
        print(f"Warning: File not found: {file_path}")
        return []

    results = []
    with open(file_path, 'r') as f:
        lines = f.readlines()

    current_test = None
    metrics = {}
    captured_score = False

    for line in lines:
        test_name = None
        if suite_type == "NPB":
            m = re_begin_npb.match(line)
            if m: test_name = m.group(1)
        elif suite_type == "UnixBench":
            m = re_begin_unix.match(line)
            if m: test_name = m.group(1)

        if test_name:
            current_test = test_name
            metrics = {}
            captured_score = False
            continue

        if re_end.match(line):
            if current_test and 'score' in metrics:
                metrics['name'] = current_test
                results.append(metrics)
            current_test = None
            continue

        if current_test:
            if not captured_score:
                if suite_type == "NPB":
                    m = re_mops.match(line)
                    if m:
                        metrics['score'] = float(m.group(1))
                        metrics['units'] = "Mop/s"
                        captured_score = True
                elif suite_type == "UnixBench":
                    m = re_unix_count.match(line)
                    if m:
                        metrics['score'] = float(m.group(1))
                        metrics['units'] = m.group(3)
                        captured_score = True

            m_inst = re_papi_inst.match(line)
            if m_inst: metrics['instructions'] = int(m_inst.group(1))
            m_cyc = re_papi_cyc.match(line)
            if m_cyc: metrics['cycles'] = int(m_cyc.group(1))
            m_ipc = re_papi_ipc.match(line)
            if m_ipc: metrics['ipc'] = float(m_ipc.group(1))

    return results

UNIX_NAME_REGEX  = re.compile(r"([A-Za-z_]+)")
UNIX_COUNT_REGEX = re.compile(r"COUNT\|([\d\.]+)\|\d+\|([A-Za-z]+)")

INST_REGEX = re.compile(r"\[PAPI\] Instructions:\s+(\d+)")

IPC_REGEX  = re.compile(r"\[PAPI\] IPC:\s+([\d\.]+)")

def parse_bitstream_name(filename):
    """Extracts parameters from the bitstream filename."""
    match = FILENAME_PATTERN.search(filename)
    if match:
        return {
            'IC_SIZE': match.group(1), 'IC_ASSOC': match.group(2),
            'DC_SIZE': match.group(3), 'DC_ASSOC': match.group(4),

        }
    return None

def parse_content_block(block_text, suite_type):
    """Parses a single benchmark block (between BEGIN_TEST and END_TEST)."""
    data = {'score': '0', 'units': '-', 'inst': '0', 'cyc': '0', 'ipc': '0'}
    
    # 1. Parse Score & Units
    if suite_type == 'NPB':
        m = NPB_SCORE_REGEX.search(block_text)
        if m:
            data['score'] = m.group(1)
            data['units'] = "Mop/s"
            
    elif suite_type == 'UnixBench':
        # Finds first occurrence of COUNT (handles Context Switching duplicates)
        m = UNIX_COUNT_REGEX.search(block_text)
        if m:
            data['score'] = m.group(1)
            data['units'] = m.group(2)

    # 2. Parse PAPI Metrics
    m_inst = INST_REGEX.search(block_text)
    m_cyc = CYC_REGEX.search(block_text)
    m_ipc = IPC_REGEX.search(block_text)

    if m_inst: data['inst'] = m_inst.group(1)
    if m_cyc:  data['cyc']  = m_cyc.group(1)
    if m_ipc:  data['ipc']  = m_ipc.group(1)
    
    return data

def process_file_content(filepath, suite_type):
    """Reads a file, splits it by test blocks, and returns a dict of results."""
    results = {}
    
    if not os.path.exists(filepath):
        # Only print once per run to avoid spam
        # print(f"[WARN] File missing: {filepath}")
        return results

    try:
        with open(filepath, 'r', errors='replace') as f:
            content = f.read()
    except Exception as e:
        print(f"[WARN] Read error {filepath}: {e}")
        return results

    # Split content by the test delimiter
    blocks = content.split('>>>BEGIN_TEST|')
    
    for block in blocks[1:]:
        lines = block.split('\n', 1)
        header = lines[0].strip()
        body = lines[1] if len(lines) > 1 else ""

        bench_name = None
        if suite_type == 'NPB':
            m = NPB_NAME_REGEX.match(header)
            if m: bench_name = m.group(1)
        elif suite_type == 'UnixBench':
            m = UNIX_NAME_REGEX.match(header)
            if m: bench_name = m.group(1)

        if bench_name:
            results[bench_name] = parse_content_block(body, suite_type)

    return results

def get_config_label(p, sweep_type):
    """Generates the label (first column) based on the sweep type."""
    if sweep_type == 'CACHE': return f"IC/DC {p['IC_SIZE']}"
    if sweep_type == 'ASSOC': return f"{p['IC_ASSOC']}w"
    if sweep_type == 'BTB':   return f"BTB{p['BTB']}"
    if sweep_type == 'BHT':   return f"BHT{p['BHT']}"
    return "Unknown"

def generate_rows(p, sweep_type, npb_data, unix_data):
    """Creates the list of CSV rows for a specific config."""
    rows = []
    config_label = get_config_label(p, sweep_type)

    # NPB Benchmarks
    for bname in ['EP', 'IS', 'CG', 'MG', 'FT', 'BT', 'SP', 'LU']:
        if bname in npb_data:
            d = npb_data[bname]
            rows.append([config_label, "NPB", bname, d['score'], d['units'], d['inst'], d['cyc'], d['ipc']])

    # UnixBench Benchmarks
    unix_order = ['Arithmetic_Overhead', 'Arithmetic_Register', 'Arithmetic_Short', 'Arithmetic_Int',

    for bname in unix_order:
        if bname in unix_data:
            d = unix_data[bname]
            rows.append([config_label, "UnixBench", bname, d['score'], d['units'], d['inst'], d['cyc'], d['ipc']])
            
    return rows

def is_error(filename):

                csv_buffers['BTB'].extend(generate_rows(p, 'BTB', npb_data, unix_data))

            # 4. BHT Sweep (Include if Baseline OR BHT changed)
            if is_baseline or (p['BHT'] != BASELINE['BHT']):
                csv_buffers['BHT'].extend(generate_rows(p, 'BHT', npb_data, unix_data))

    # --- WRITE TO FILES ---
    header_row = ["config", "suite", "name", "score", "units", "instructions", "cycles", "ipc"]
    
    for key, filename in OUTPUT_FILES.items():
        if csv_buffers[key]:
            with open(filename, 'w', newline='') as f:
                writer = csv.writer(f)
                writer.writerow(header_row)
                writer.writerows(csv_buffers[key])
            print(f"Created {filename}: {len(csv_buffers[key])} rows")
        else:
            print(f"Warning: No data found for {filename}")

    print("Done!")
